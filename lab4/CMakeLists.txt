add_library(lab4_core INTERFACE)
target_include_directories(lab4_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(lab4_core INTERFACE cxx_std_20)
if(MSVC)
    target_compile_options(lab4_core INTERFACE /W4)
else()
    target_compile_options(lab4_core INTERFACE -Wall -Wextra -Wpedantic)
endif()

option(LAB4_ENABLE_COVERAGE "Instrument lab4 with LLVM coverage" OFF)
option(LAB4_ENABLE_SANITIZERS "Check lab4 with ASan and UBSan" OFF)
if(LAB4_ENABLE_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "LAB4_ENABLE_COVERAGE requires Clang or AppleClang")
    endif()
    target_compile_options(lab4_core INTERFACE -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(lab4_core INTERFACE -fprofile-instr-generate)
endif()
if(LAB4_ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "LAB4_ENABLE_SANITIZERS requires Clang or GCC")
    endif()
    target_compile_options(lab4_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lab4_core INTERFACE -fsanitize=address,undefined)
endif()

add_library(lab4_ui STATIC app/Console.cpp)
target_include_directories(lab4_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/app")
target_link_libraries(lab4_ui PUBLIC lab4_core)
add_executable(lab4_cli app/main.cpp)
target_link_libraries(lab4_cli PRIVATE lab4_ui)
install(TARGETS lab4_cli RUNTIME DESTINATION bin COMPONENT lab4)
install(FILES README.md DESTINATION share/ppois_lab4 COMPONENT lab4)
install(FILES examples/demo.input DESTINATION share/ppois_lab4/examples COMPONENT lab4)

find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
    configure_file(Doxyfile.in Doxyfile @ONLY)
    add_custom_target(lab4_docs
        COMMAND "${DOXYGEN_EXECUTABLE}" "${CMAKE_CURRENT_BINARY_DIR}/Doxyfile"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        COMMENT "Generating sorting and graph HTML documentation" VERBATIM)
endif()
